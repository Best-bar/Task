const AUTH_USERS_PROPERTY = 'AUTH_USERS';
const AUTH_SECRET_PROPERTY = 'AUTH_SECRET';
const AUTH_TOKEN_DAYS = 365;

function bytesToHex_(bytes) {
  return bytes.map(function(b) {
    var v = b < 0 ? b + 256 : b;
    return ('0' + v.toString(16)).slice(-2);
  }).join('');
}

function getAuthSecret_() {
  var props = PropertiesService.getScriptProperties();
  var secret = props.getProperty(AUTH_SECRET_PROPERTY);
  if (!secret) {
    secret = Utilities.getUuid() + ':' + Utilities.getUuid() + ':' + new Date().getTime();
    props.setProperty(AUTH_SECRET_PROPERTY, secret);
  }
  return secret;
}

function hashPassword_(password, salt) {
  var bytes = Utilities.computeDigest(
    Utilities.DigestAlgorithm.SHA_256,
    String(password) + '|' + String(salt),
    Utilities.Charset.UTF_8
  );
  return bytesToHex_(bytes);
}

function getAuthUsers_() {
  var raw = PropertiesService.getScriptProperties().getProperty(AUTH_USERS_PROPERTY);
  if (!raw) return [];
  try {
    var parsed = JSON.parse(raw);
    return Array.isArray(parsed) ? parsed : [];
  } catch (e) {
    Logger.log('Ошибка чтения AUTH_USERS: ' + e.toString());
    return [];
  }
}

// Запустите эту функцию вручную из редактора Apps Script после замены паролей.
function setupAuthorization() {
  setAuthUser_('admin', 'admin777', 'admin', 'Администратор');
  setAuthUser_('requester', 'CHANGE_REQUESTER_PASSWORD', 'requester', 'Подающий заявку');

  getAuthSecret_();
  Logger.log('Авторизация настроена. Используйте свои логины и пароли.');
}

// Вспомогательная функция настройки пользователя. Выполняется только из редактора Apps Script.
function setAuthUser_(login, password, role, name) {
  var allowedRoles = ['admin', 'requester'];
  login = String(login || '').trim().toLowerCase();
  password = String(password || '');
  role = String(role || '').trim().toLowerCase();
  name = String(name || login).trim();

  if (!login || !password) throw new Error('Для пользователя необходимо указать логин и пароль.');
  if (allowedRoles.indexOf(role) === -1) throw new Error('Недопустимая роль: ' + role);

  var users = getAuthUsers_();
  var salt = Utilities.getUuid();
  var user = {
    login: login,
    name: name,
    role: role,
    salt: salt,
    passwordHash: hashPassword_(password, salt),
    enabled: true
  };

  var replaced = false;
  for (var i = 0; i < users.length; i++) {
    if (String(users[i].login || '').toLowerCase() === login) {
      users[i] = user;
      replaced = true;
      break;
    }
  }
  if (!replaced) users.push(user);

  PropertiesService.getScriptProperties().setProperty(AUTH_USERS_PROPERTY, JSON.stringify(users));
}

function makeAuthToken_(user) {
  var now = Math.floor(Date.now() / 1000);
  var payload = {
    v: 1,
    login: user.login,
    name: user.name,
    role: user.role,
    iat: now,
    exp: now + (AUTH_TOKEN_DAYS * 24 * 60 * 60)
  };

  var payloadText = Utilities.base64EncodeWebSafe(JSON.stringify(payload));
  var signatureBytes = Utilities.computeHmacSha256Signature(payloadText, getAuthSecret_());
  var signature = Utilities.base64EncodeWebSafe(signatureBytes);
  return payloadText + '.' + signature;
}

function validateAuthToken_(token) {
  if (!token || typeof token !== 'string') return null;

  try {
    var parts = token.split('.');
    if (parts.length !== 2) return null;

    var payloadText = Utilities.newBlob(Utilities.base64DecodeWebSafe(parts[0])).getDataAsString();
    var payload = JSON.parse(payloadText);
    if (!payload || payload.v !== 1 || !payload.login || !payload.role || !payload.exp) return null;
    if (Number(payload.exp) < Math.floor(Date.now() / 1000)) return null;

    var expectedBytes = Utilities.computeHmacSha256Signature(parts[0], getAuthSecret_());
    var expectedSignature = Utilities.base64EncodeWebSafe(expectedBytes);
    if (expectedSignature !== parts[1]) return null;

    var users = getAuthUsers_();
    var user = users.find(function(u) {
      return String(u.login || '').toLowerCase() === String(payload.login).toLowerCase() && u.enabled !== false;
    });
    if (!user) return null;
    if (String(user.role) !== String(payload.role)) return null;

    return {
      login: user.login,
      name: user.name || user.login,
      role: user.role
    };
  } catch (e) {
    return null;
  }
}

function authRequire_(token, allowedRoles) {
  var user = validateAuthToken_(token);
  if (!user) throw new Error('Сессия авторизации недействительна или истекла. Войдите в приложение заново.');
  if (allowedRoles && allowedRoles.indexOf(user.role) === -1) {
    throw new Error('Недостаточно прав для выполнения этой операции.');
  }
  return user;
}

function authLogin(login, password) {
  try {
    login = String(login || '').trim().toLowerCase();
    password = String(password || '');

    if (!login || !password) {
      return { success: false, error: 'Введите логин и пароль.' };
    }

    var users = getAuthUsers_();
    if (!users.length) {
      return { success: false, error: 'Авторизация еще не настроена. Запустите setupAuthorization_() в редакторе Apps Script.' };
    }

    var user = users.find(function(u) {
      return String(u.login || '').toLowerCase() === login && u.enabled !== false;
    });

    if (!user || hashPassword_(password, user.salt) !== user.passwordHash) {
      return { success: false, error: 'Неверный логин или пароль.' };
    }

    var publicUser = {
      login: user.login,
      name: user.name || user.login,
      role: user.role
    };

    return {
      success: true,
      token: makeAuthToken_(user),
      user: publicUser
    };
  } catch (e) {
    Logger.log('Ошибка authLogin: ' + e.toString());
    return { success: false, error: e.toString() };
  }
}

function authCheck(token) {
  var user = validateAuthToken_(token);
  if (!user) return { success: false };
  return { success: true, user: user };
}

// Подача заявок и просмотр задач остаются доступны как раньше.
function getRequesterAuthToken() {
  var users = getAuthUsers_();
  var user = users.find(function(u) {
    return String(u.login || '').toLowerCase() === 'requester' && u.enabled !== false;
  });

  if (!user) {
    var randomPassword = Utilities.getUuid() + Utilities.getUuid();
    setAuthUser_('requester', randomPassword, 'requester', 'Пользователь');
    users = getAuthUsers_();
    user = users.find(function(u) {
      return String(u.login || '').toLowerCase() === 'requester' && u.enabled !== false;
    });
  }

  return {
    success: true,
    token: makeAuthToken_(user),
    user: { login: user.login, name: user.name || 'Пользователь', role: 'requester' }
  };
}

// Совместимость с текущим интерфейсом: Index1 вызывает эту функцию
// при нажатии на надпись «Пользователь».
function checkAdminPassword(password) {
  try {
    password = String(password || '');
    if (!password) return false;
    var result = authLogin('admin', password);
    if (result && result.success && result.user && result.user.role === 'admin') {
      return { success: true, token: result.token, user: result.user };
    }
    return { success: false };
  } catch (e) {
    Logger.log('Ошибка checkAdminPassword: ' + e.toString());
    return false;
  }
}

    sheet.getRange(1, 1, 1, TASKS_HEADERS.length).setValues([TASKS_HEADERS]);
  }
  return sheet;
}

/**
 * Одноразовая миграция старых листов-баров из ТЕКУЩЕЙ (скопированной)
 * Google Таблицы в общий лист "Задачи".
 *
 * ВАЖНО:
 * - функция работает только с таблицей, к которой привязан ЭТОТ новый Apps Script;
 * - старые листы не удаляются и не изменяются;
 * - переносит только листы, названия которых есть в FIXED_BARS;
 * - в колонку "Бар" записывается каноническое название из FIXED_BARS;
 * - повторный запуск блокируется отдельным флагом миграции этой копии.
 *
 * Если в старой таблице была одна общая копия с историей, это именно тот
 * вариант, который нужен для тестового проекта.
 */
function migrateExistingBarSheetsToMaster() {
  var lock = LockService.getScriptLock();
  lock.waitLock(30000);

  try {
    var props = PropertiesService.getScriptProperties();
    var migrationFlag = 'MASTER_MIGRATION_FROM_CURRENT_COPY_DONE';

    if (props.getProperty(migrationFlag) === '1') {
      return {
        success: true,
        message: 'Перенос из текущей копии уже выполнялся. Старые листы не изменялись.'
      };
    }

    // Источник и приёмник — одна и та же НОВАЯ скопированная таблица.
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    if (!ss) throw new Error('Не удалось определить текущую Google Таблицу.');

    var master = ss.getSheetByName(TASKS_SHEET_NAME);
    if (!master) {
      master = ss.insertSheet(TASKS_SHEET_NAME);
      master.getRange(1, 1, 1, TASKS_HEADERS.length).setValues([TASKS_HEADERS]);
    } else if (master.getLastColumn() < TASKS_HEADERS.length) {
      master.getRange(1, 1, 1, TASKS_HEADERS.length).setValues([TASKS_HEADERS]);
    }

    // Создаём соответствие "название листа -> название бара из FIXED_BARS".
    // Пробелы по краям и регистр не мешают сопоставлению.
    var barMap = {};
    FIXED_BARS.forEach(function(barName) {
      var canonical = String(barName || '').trim();
      if (canonical) barMap[canonical.toLowerCase()] = canonical;
    });

    var sourceSheets = ss.getSheets();
    var copied = 0;
    var skipped = [];
    var migratedBars = [];

    sourceSheets.forEach(function(sheet) {
      var sheetName = String(sheet.getName() || '').trim();
      if (!sheetName || sheetName === TASKS_SHEET_NAME) return;

      var canonicalBarName = barMap[sheetName.toLowerCase()];
      if (!canonicalBarName) return;

      var data = sheet.getDataRange().getValues();
      if (data.length <= 1) {
        skipped.push(sheetName + ' — нет строк с задачами');
        return;
      }

      var rows = data.slice(1)
        .filter(function(row) {
          return row[0] || row[3];
        })
        .map(function(row) {
          // Старые листы имеют 10 основных колонок.
          // В новую структуру добавляем Бар и Регион.
          return [
            row[0] || '',  // Дата
            row[1] || '',  // Тип заявки
            row[2] || '',  // Инициатор
            row[3] || '',  // Что сделать
            row[4] || '',  // Сроки устранения
            row[5] || '',  // Фото заявки
            row[6] || '',  // Комментарии
            row[7] || '',  // Дата выполнения
            row[8] || '',  // Результат выполнения
            row[9] || '',  // Фото результата
            canonicalBarName, // Бар
            getBarRegion_(canonicalBarName) // Регион
          ];
        });

      if (!rows.length) {
        skipped.push(sheetName + ' — подходящих задач нет');
        return;
      }

      var firstRow = master.getLastRow() + 1;
      master.getRange(firstRow, 1, rows.length, TASKS_HEADERS.length).setValues(rows);
      copied += rows.length;
      migratedBars.push(canonicalBarName + ': ' + rows.length);
    });

    // Ставим флаг только после успешного завершения переноса.
    props.setProperty(migrationFlag, '1');

    // Сбрасываем кэш, чтобы история появилась сразу после миграции.
    clearTasksCache();

    return {
      success: true,
      copied: copied,
      bars: migratedBars,
      skipped: skipped,
      message: 'Перенос из текущей копии завершён. Старые листы не изменялись.'
    };

  } catch (e) {
    Logger.log('Ошибка миграции из текущей копии: ' + e.toString());
    return { success: false, error: e.toString() };
  } finally {
    lock.releaseLock();
  }
}

/**
 * Универсальная функция для сохранения Base64 файла в конкретную папку на Google Диске по ID
 */
function saveFileToTargetFolder_(fileData) {
  if (!fileData || !fileData.base64) return "";

  try {
    var folder = DriveApp.getFolderById(TARGET_FOLDER_ID);

    var contentType = fileData.type || "application/octet-stream";
    var fileName = fileData.name || (contentType === "image/heic" ? "photo.heic" : "photo");

    var lowerName = fileName.toLowerCase();
    if (lowerName.endsWith(".heic")) contentType = "image/heic";
    if (lowerName.endsWith(".heif")) contentType = "image/heif";

    var base64Part = fileData.base64.indexOf(",") >= 0
      ? fileData.base64.split(",")[1]
      : fileData.base64;
    var decode = Utilities.base64Decode(base64Part);
    var blob = Utilities.newBlob(decode, contentType, fileName);
    var file = folder.createFile(blob);

    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);

    var fileId = file.getId();
    return "https://docs.google.com/uc?export=view&id=" + fileId;

  } catch (e) {
    Logger.log("Ошибка при сохранении файла на Диск: " + e.toString());
    throw new Error("Не удалось сохранить файл на Google Диск: " + e.toString());
  }
}

// Загрузка одного уже обработанного изображения.
function uploadSingleFile(fileData, authToken) {
  try {
    authRequire_(authToken, ['admin', 'requester']);
    return saveFileToTargetFolder_(fileData);
  } catch (e) {
    Logger.log('Ошибка uploadSingleFile: ' + e.toString());
    throw new Error(e.toString());
  }
}


// Одноразово назначает регионы уже перенесённым задачам в листе "Задачи".
// Запускать после изменения REGION_BARS, если миграция уже была выполнена.
function assignRegionsToExistingTasks_() {
  var lock = LockService.getScriptLock();
  lock.waitLock(30000);
  try {
    var sheet = getTasksSheet_();
    var lastRow = sheet.getLastRow();
    if (lastRow < 2) return { success: true, updated: 0 };

    var values = sheet.getRange(2, 11, lastRow - 1, 2).getValues();
    var updated = 0;
    values.forEach(function(row) {
      var barName = String(row[0] || '').trim();
      if (!barName) return;
      var region = getBarRegion_(barName);
      if (row[1] !== region) {
        row[1] = region;
        updated++;
      }
    });

    if (values.length) sheet.getRange(2, 11, values.length, 2).setValues(values); // This is perfectly fine, it writes BOTH columns K and L back.
    clearTasksCache();
    return { success: true, updated: updated, message: 'Регионы назначены для существующих задач.' };
  } catch (e) {
    Logger.log('Ошибка assignRegionsToExistingTasks_: ' + e.toString());
    return { success: false, error: e.toString() };
  } finally {
    lock.releaseLock();
  }
}

// ЭКРАН 2: Создание новой заявки (Структура из 10 столбцов)
// ==========================================================
function processForm(payload, authToken) {
  try {
    authRequire_(authToken, ['admin', 'requester']);
    if (!payload || !FIXED_BARS.includes(payload.barName)) {
      throw new Error('Неизвестный бар: ' + (payload && payload.barName ? payload.barName : ''));
    }

    const sheet = getTasksSheet_();
    const timestamp = new Date();
    const photoUrls = [];

    const deadlineDate = parseRussianDate_(payload.deadline);
    if (!deadlineDate || isNaN(deadlineDate.getTime())) {
      throw new Error('Некорректный срок устранения: ' + payload.deadline);
    }

    if (payload.files && payload.files.length > 0) {
      for (var i = 0; i < payload.files.length; i++) {
        if (typeof payload.files[i] === 'string') {
          if (payload.files[i]) photoUrls.push(payload.files[i]);
        } else if (payload.files[i] && payload.files[i].base64) {
          var url = saveFileToTargetFolder_(payload.files[i]);
          if (url) photoUrls.push(url);
        }
      }
    }

    let finalTaskText = payload.taskText;
    if (payload.urgent) finalTaskText = '⚠️ [СРОЧНО] ' + finalTaskText;

    const lock = LockService.getScriptLock();
    lock.waitLock(15000);
    try {
      sheet.appendRow([
        timestamp,
      payload.requestType,
      payload.initiator,
      finalTaskText,
      deadlineDate,
      photoUrls.join(', '),
      '',
      '',
      '',
      '',
      payload.barName,
        getBarRegion_(payload.barName)
      ]);
    } finally {
      lock.releaseLock();
    }

    sendTelegramNotification_(payload);
    clearTasksCache();
    return { success: true };
  } catch (error) {
    Logger.log('Ошибка в processForm: ' + error.toString());
    return { success: false, error: error.toString() };
  }
}

function getBarTasks(barName, region, authToken) {
  try {
    authRequire_(authToken, ['admin', 'requester']);
    if (barName !== 'ALL' && !FIXED_BARS.includes(barName)) {
      return { success: false, error: 'Неизвестный бар.' };
    }
    const version = getCacheVersion_();
    const cacheKey = 'tasks_master_' + (region || 'all') + '_' + barName + '_' + version;
    const cache = CacheService.getScriptCache();
    const cached = cache.get(cacheKey);
    if (cached) {
      return { success: true, tasks: JSON.parse(cached) };
    }

    const sheet = getTasksSheet_();
    let tasks = fetchTasksFromSheet_(sheet, TASKS_SHEET_NAME);
    if (barName !== 'ALL') tasks = tasks.filter(t => t.barName === barName);
    if (region) tasks = tasks.filter(t => {
      var taskRegion = String(t.region || '').toLowerCase();
      if (!taskRegion && t.barName) {
        taskRegion = getBarRegion_(t.barName).toLowerCase();
      }
      var requestedRegion = String(region).toLowerCase();
      return taskRegion === requestedRegion;
    });
    tasks.reverse();
    try { cache.put(cacheKey, JSON.stringify(tasks), 120); } catch (e) {}
    return { success: true, tasks: tasks };
  } catch (error) {
    return { success: false, error: error.toString() };
  }
}

function fetchTasksFromSheet_(sheet, sheetName) {
  const data = sheet.getDataRange().getValues();
  if (data.length <= 1) return [];

  const tasks = [];
  const now = new Date();
  for (let i = 1; i < data.length; i++) {
    const row = data[i];
    if (!row[0] && !row[3]) continue;

    let pUrls = row[5] ? row[5].toString().split(',').map(u => u.trim()).filter(u => u !== '') : [];
    const rawText = row[3] ? row[3].toString() : '';
    const isUrgentText = rawText.includes('⚠️ [СРОЧНО]');

    // Handle the creation date (row[0]) correctly
    const creationDate = row[0] instanceof Date ? row[0] : parseRussianDate_(row[0]);
    const creationDateStr = creationDate
      ? Utilities.formatDate(creationDate, Session.getScriptTimeZone(), 'dd.MM.yyyy HH:mm')
      : (row[0] ? row[0].toString() : '');

    const deadlineDate = row[4] instanceof Date ? row[4] : parseRussianDate_(row[4]);
    const deadlineStr = deadlineDate
      ? Utilities.formatDate(deadlineDate, Session.getScriptTimeZone(), 'dd.MM.yyyy HH:mm')
      : (row[4] ? row[4].toString() : '');
    const resultText = row[8] ? row[8].toString() : '';
    const commentsText = row[6] ? row[6].toString() : '';
    const isInWork = commentsText.includes('[В работе]');
    const hasResult = resultText.trim() !== '';
    const isOverdue = !hasResult && deadlineDate && deadlineDate < now;

    const fixUrl = function(url) {
      if (url.includes('id=')) return url;
      if (url.includes('drive.google.com/file/d/')) {
        var id = url.split('/file/d/')[1].split('/')[0];
        return 'https://docs.google.com/uc?export=view&id=' + id;
      }
      return url;
    };
    const fixedPhotos = pUrls.map(fixUrl);
    let fixedResultPhoto = '';
    if (row[9]) fixedResultPhoto = row[9].toString().split(',').map(u => fixUrl(u.trim())).filter(u => u !== '').join(', ');

    tasks.push({
      rowIdx: i + 1,
      barName: row[10] ? row[10].toString() : '',
      region: row[11] ? row[11].toString() : '',
      date: creationDateStr,
      type: row[1] ? row[1].toString() : 'Общее',
      initiator: row[2] ? row[2].toString() : 'Не указан',
      text: rawText,
      deadline: deadlineStr,
      photos: fixedPhotos,
      comment: commentsText,
      doneDate: row[7] instanceof Date ? Utilities.formatDate(row[7], Session.getScriptTimeZone(), 'dd.MM.yyyy HH:mm') : (row[7] ? row[7].toString() : ''),
      result: resultText,
      resultPhoto: fixedResultPhoto,
      urgent: isUrgentText,
      overdue: isOverdue,
      inWork: isInWork
    });
  }
  return tasks;
}

function adminUpdateTask(taskRef, updates, filesData, authToken) {
  try {
    authRequire_(authToken, ['admin']);
    if (!taskRef || !taskRef.barName || !taskRef.rowIdx) return { success:false, error:'Не указана исходная задача.' };
    updates = updates || {};
    const sheet = getTaskSheetAndCheckRow_(taskRef.rowIdx);
    if (String(sheet.getRange(taskRef.rowIdx, 11).getValue()) !== String(taskRef.barName)) return { success:false, error:'Исходная задача не найдена.' };
    const oldRow = sheet.getRange(taskRef.rowIdx, 1, 1, 12).getValues()[0];
    const targetBarName = updates.barName || taskRef.barName;
    if (!FIXED_BARS.includes(targetBarName)) return { success:false, error:'Неизвестный бар.' };

    const deadlineDate = parseRussianDate_(updates.deadline);
    if (!deadlineDate || isNaN(deadlineDate.getTime())) return { success:false, error:'Некорректный срок устранения: ' + (updates.deadline || '') };
    let taskText = String(updates.taskText || '').replace(/^\s*⚠️\s*\[СРОЧНО\]\s*/i, '');
    const finalTaskText = updates.urgent ? '⚠️ [СРОЧНО] ' + taskText : taskText;

    let photos = oldRow[5] ? oldRow[5].toString() : '';
    if (Array.isArray(filesData)) {
      const addedUrls = [];
      filesData.forEach(function(f){ if (f && f.base64) { const u=saveFileToTargetFolder_(f); if(u) addedUrls.push(u); } });
      if (addedUrls.length) photos = photos ? photos + ', ' + addedUrls.join(', ') : addedUrls.join(', ');
    }

    const newRow = [oldRow[0], updates.requestType || oldRow[1], updates.initiator || oldRow[2], finalTaskText, deadlineDate, photos, oldRow[6], oldRow[7], oldRow[8], oldRow[9], targetBarName, updates.region || oldRow[11] || taskRef.region || ''];
    sheet.getRange(taskRef.rowIdx, 1, 1, 12).setValues([newRow]);
    clearTasksCache();
    return { success:true, region:newRow[11], barName:targetBarName, rowIdx:taskRef.rowIdx };
  } catch (error) { Logger.log('Ошибка adminUpdateTask: '+error.toString()); return {success:false,error:error.toString()}; }
}

function parseRussianDate_(dateStr) {
  try {
    if (!dateStr) return null;
    if (dateStr instanceof Date) return new Date(dateStr.getTime());

    const value = dateStr.toString().trim();
    if (!value) return null;

    const parts = value.split(" ");
    const dParts = parts[0].split(".");

    if (dParts.length === 3) {
      const day = parseInt(dParts[0], 10);
      const month = parseInt(dParts[1], 10) - 1;
      const year = parseInt(dParts[2], 10);

      if ([day, month, year].some(Number.isNaN)) return null;

      let hour = 0;
      let minute = 0;

      if (parts[1]) {
        const tParts = parts[1].split(":");
        hour = parseInt(tParts[0], 10) || 0;
        minute = parseInt(tParts[1], 10) || 0;
      }

      return new Date(year, month, day, hour, minute, 0, 0);
    }

    const isoDate = new Date(value);
    return isNaN(isoDate.getTime()) ? null : isoDate;
  } catch (e) {
    return null;
  }
}

function sendTelegramNotification_(payload) {
  try {
    if (!TELEGRAM_BOT_TOKEN) return;

    let targetChatId = "";

    const reqTypeLower = (payload.requestType || "").toLowerCase();
    if (reqTypeLower.indexOf("тех") !== -1) {
      targetChatId = TECH_EQUIPMENT_SPECIALIST_CHAT_ID;
    } else {
      targetChatId = BAR_TELEGRAM_RECIPIENTS[payload.barName] || BAR_TELEGRAM_RECIPIENTS["default"];
    }

    if (!targetChatId) {
      targetChatId = TECH_EQUIPMENT_SPECIALIST_CHAT_ID;
    }

    if (!targetChatId) return;

    const regionText = (payload.region === "regions" || payload.region === "Регионы") ? "Регионы" : "Москва";
    const urgentBadge = payload.urgent ? "🚨 <b>[СРОЧНАЯ ЗАДАЧА]</b>\n" : "";

    const text = `${urgentBadge}` +
                 `🛠️ <b>Новая заявка: ${payload.barName} (${regionText})</b>\n\n` +
                 `📍 <b>Регион:</b> ${regionText}\n` +
                 `🏷️ <b>Тип:</b> ${payload.requestType}\n` +
                 `👤 <b>Инициатор:</b> ${payload.initiator}\n` +
                 `⏰ <b>Дедлайн:</b> ${payload.deadline}\n\n` +
                 `📝 <b>Что сделать:</b>\n${payload.taskText}\n\n` +
                 `🔗 <a href="${ScriptApp.getService().getUrl()}">Открыть приложение</a>`;

    const url = `https://api.telegram.org/bot${TELEGRAM_BOT_TOKEN}/sendMessage`;

    UrlFetchApp.fetch(url, {
      method: "post",
      contentType: "application/json",
      payload: JSON.stringify({
        chat_id: targetChatId,
        text: text,
        parse_mode: "HTML",
        disable_web_page_preview: true
      }),
      muteHttpExceptions: true
    });

  } catch (err) {
    Logger.log("Ошибка отправки в Telegram: " + err.toString());
  }
}
