// ЭКРАН 1: Получение списка листов с серверным кэшированием (CacheService)
// ==========================================================
function getBars(region) {
  try {
    const regionKey = (region === "regions" || region === "Регионы") ? "regions" : "moscow";
    const cacheKey = "bars_list_" + regionKey;
    const cache = CacheService.getScriptCache();
    const cached = cache.get(cacheKey);

    if (cached) {
      try {
        const parsed = JSON.parse(cached);
        if (Array.isArray(parsed) && parsed.length > 0) {
          return { success: true, bars: parsed };
        }
      } catch (e) {}
    }

    const ss = getSpreadsheetByRegion(region);
    const sheets = ss.getSheets();
    const barNames = [];
    
    sheets.forEach(sheet => {
      const name = sheet.getName();
      if (name !== "Справочник" && name !== "Настройки" && name !== "Шаблон") {
        barNames.push(name);
      }
    });
    
    if (barNames.length === 0) {
      return { success: true, bars: ["Тестовый Бар"] };
    }

    // Сохраняем в кэш на 30 минут (1800 секунд)
    try {
      cache.put(cacheKey, JSON.stringify(barNames), 1800);
    } catch (e) {}
    
    return { success: true, bars: barNames };
  } catch (error) {
    Logger.log("Ошибка в getBars: " + error.toString());
    return { success: false, error: error.toString(), bars: [] };
  }
}

/**
 * Универсальная функция для сохранения Base64 файла в папку на Google Диске
 */
function saveFileToTargetFolder(fileData) {
  if (!fileData || !fileData.base64) return "";
  
  try {
    var folder = DriveApp.getFolderById(TARGET_FOLDER_ID);
    
    var contentType = fileData.type || "application/octet-stream";
    var fileName = fileData.name || (contentType === "image/heic" ? "photo.heic" : "photo");

    var lowerName = fileName.toLowerCase();
    if (lowerName.endsWith(".heic")) contentType = "image/heic";
    if (lowerName.endsWith(".heif")) contentType = "image/heif";

    var base64Part = fileData.base64.indexOf(",") >= 0
      ? fileData.base64.split(",")[1]
      : fileData.base64;
    var decode = Utilities.base64Decode(base64Part);
    var blob = Utilities.newBlob(decode, contentType, fileName);
    var file = folder.createFile(blob);
    
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    
    var fileId = file.getId();
    return "https://docs.google.com/uc?export=view&id=" + fileId;
    
  } catch (e) {
    Logger.log("Ошибка при сохранении файла на Диск: " + e.toString());
    throw new Error("Не удалось сохранить файл на Google Диск: " + e.toString());
  }
}

/**
 * Загрузка одного файла из браузера
 */
function uploadSingleFile(fileData) {
  try {
    return saveFileToTargetFolder(fileData);
  } catch (error) {
    return "";
  }
}

// ЭКРАН 2: Создание новой заявки
// ==========================================================
function processForm(payload) {
  try {
    const ss = getSpreadsheetByRegion(payload ? payload.region : null);
    let sheet = ss.getSheetByName(payload.barName);
    
    if (!sheet) {
      sheet = ss.insertSheet(payload.barName);
      sheet.appendRow(["Дата", "Тип заявки", "Инициатор", "Что сделать", "Сроки устранения", "Фото заявки", "Комментарии", "Дата выполнения", "Результат выполнения", "Фото результата"]);
      
      // Сбрасываем кэш баров при создании нового листа
      try {
        const regionKey = (payload && (payload.region === "regions" || payload.region === "Регионы")) ? "regions" : "moscow";
        CacheService.getScriptCache().remove("bars_list_" + regionKey);
      } catch (e) {}
    }

    const timestamp = new Date();
    const photoUrls = [];

    const deadlineDate = parseRussianDate(payload.deadline);
    if (!deadlineDate || isNaN(deadlineDate.getTime())) {
      throw new Error("Некорректный срок устранения: " + payload.deadline);
    }

    if (payload.files && payload.files.length > 0) {
      for (var i = 0; i < payload.files.length; i++) {
        if (payload.files[i] && payload.files[i].base64) {
          var url = saveFileToTargetFolder(payload.files[i]);
          if (url) photoUrls.push(url);
        } else if (typeof payload.files[i] === 'string' && payload.files[i].startsWith('http')) {
          // Если переданы уже готовые ссылки
          photoUrls.push(payload.files[i]);
        }
      }
    }

    const photosString = photoUrls.join(", ");
    
    let finalTaskText = payload.taskText;
    if (payload.urgent) {
      finalTaskText = "⚠️ [СРОЧНО] " + finalTaskText;
    }

    sheet.appendRow([
      timestamp,           // А (1) - Дата подачи
      payload.requestType, // Б (2) - Тип заявки
      payload.initiator,   // В (3) - Инициатор
      finalTaskText,       // Г (4) - Что сделать
      deadlineDate,        // Д (5) - Сроки устранения
      photosString,        // Е (6) - Фото заявки
      "",                  // Ж (7) - Комментарии
      "",                  // З (8) - Дата выполнения
      "",                  // И (9) - Результат
      ""                   // Й (10) - Фото результата
    ]);

    sendTelegramNotification(payload);
    clearTasksCache(payload.region);

    return { success: true };
  } catch (error) {
    Logger.log("Ошибка в processForm: " + error.toString());
    return { success: false, error: error.toString() };
  }
}

// ЭКРАН 3: Получение всех задач
// ==========================================================
function getBarTasks(barName, region) {
  try {
    const ss = getSpreadsheetByRegion(region);
    let allTasks = [];

    if (barName === "ALL") {
      const barsRes = getBars(region);
      const barsList = (barsRes && barsRes.bars) ? barsRes.bars : [];
      
      barsList.forEach(bName => {
        const sheet = ss.getSheetByName(bName);
        if (sheet) {
          const tasksFromBar = fetchTasksFromSheet(sheet, bName);
          allTasks = allTasks.concat(tasksFromBar);
        }
      });
    } else {
      const sheet = ss.getSheetByName(barName);
      if (sheet) {
        allTasks = fetchTasksFromSheet(sheet, barName);
      }
    }

    allTasks.reverse();
    return { success: true, tasks: allTasks };
  } catch (error) {
    return { success: false, error: error.toString() };
  }
}

function fetchTasksFromSheet(sheet, sheetName) {
  const data = sheet.getDataRange().getValues();
  if (data.length <= 1) return [];

  const tasks = [];
  const now = new Date();

  for (let i = 1; i < data.length; i++) {
    const row = data[i];
    if (!row[0] && !row[3]) continue;

    let pUrls = [];
    if (row[5]) {
      pUrls = row[5].toString().split(",").map(u => u.trim()).filter(u => u !== "");
    }

    const rawText = row[3] ? row[3].toString() : "";
    const isUrgentText = rawText.includes("⚠️️ [СРОЧНО]");
    
    const deadlineDate = row[4] instanceof Date ? row[4] : parseRussianDate(row[4]);
    const deadlineStr = deadlineDate
      ? Utilities.formatDate(deadlineDate, Session.getScriptTimeZone(), "dd.MM.yyyy HH:mm")
      : (row[4] ? row[4].toString() : "");
    const resultText = row[8] ? row[8].toString() : "";  
    const hasResult = resultText.trim() !== "";

    const commentsText = row[6] ? row[6].toString() : "";
    const isInWork = commentsText.includes("[В работе]");

    let isOverdue = false;
    if (!hasResult && deadlineDate && deadlineDate < now) {
      isOverdue = true;
    }

    const fixUrl = function(url) {
      if (url.includes("id=")) return url;
      if (url.includes("drive.google.com/file/d/")) {
        var id = url.split("/file/d/")[1].split("/")[0];
        return "https://docs.google.com/uc?export=view&id=" + id;
      }
      return url;
    };

    const fixedPhotos = pUrls.map(fixUrl);
    
    let fixedResultPhoto = "";
    if (row[9]) {
      fixedResultPhoto = row[9].toString().split(",")
        .map(u => fixUrl(u.trim()))
        .filter(u => u !== "")
        .join(", ");
    }

    tasks.push({
      rowIdx: i + 1,
      barName: sheetName,
      date: row[0] ? row[0].toString() : "",
      type: row[1] ? row[1].toString() : "Общее",
      initiator: row[2] ? row[2].toString() : "Не указан",
      text: rawText,
      deadline: deadlineStr,
      photos: fixedPhotos,
      comment: commentsText,
      doneDate: row[7] instanceof Date
        ? Utilities.formatDate(row[7], Session.getScriptTimeZone(), "dd.MM.yyyy HH:mm")
        : (row[7] ? row[7].toString() : ""),
      result: resultText,
      resultPhoto: fixedResultPhoto,
      urgent: isUrgentText,
      overdue: isOverdue,
      inWork: isInWork
    });
  }

  return tasks;
}

function takeTaskInWork(barName, rowIdx, region) {
  try {
    const ss = getSpreadsheetByRegion(region);
    const sheet = ss.getSheetByName(barName);
    if (!sheet) return { success: false, error: "Лист не найден" };

    const cell = sheet.getRange(rowIdx, 7);
    const currentComment = cell.getValue().toString();
    
    if (currentComment.includes("[В работе]")) {
      return { success: true, alreadyInWork: true, updatedComment: currentComment };
    }

    const timestamp = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "dd.MM.yyyy HH:mm");
    const formattedInWork = "[" + timestamp + "]: [В работе]";
    
    let finalComment = currentComment.trim() === "" ? formattedInWork : currentComment + "\n" + formattedInWork;
    
    cell.setValue(finalComment);
    clearTasksCache(region);
    return { success: true, updatedComment: finalComment };
  } catch (error) {
    return { success: false, error: error.toString() };
  }
}

function parseRussianDate(dateStr) {
  try {
    if (!dateStr) return null;
    if (dateStr instanceof Date) return new Date(dateStr.getTime());

    const value = dateStr.toString().trim();
    if (!value) return null;

    const parts = value.split(" ");
    const dParts = parts[0].split(".");

    if (dParts.length === 3) {
      const day = parseInt(dParts[0], 10);
      const month = parseInt(dParts[1], 10) - 1;
      const year = parseInt(dParts[2], 10);

      if ([day, month, year].some(Number.isNaN)) return null;

      let hour = 0;
      let minute = 0;

      if (parts[1]) {
        const tParts = parts[1].split(":");
        hour = parseInt(tParts[0], 10) || 0;
        minute = parseInt(tParts[1], 10) || 0;
      }

      return new Date(year, month, day, hour, minute, 0, 0);
    }

    const isoDate = new Date(value);
    return isNaN(isoDate.getTime()) ? null : isoDate;
  } catch (e) {
    return null;
  }
}

/**
 * Сбор сводной статистики и задач тех. оборудования по всем барам
 */
function getTechEquipmentDashboardData(region) {
  const cacheKey = "tech_dashboard_data_" + (region === "regions" || region === "Регионы" ? "regions" : "moscow");
  const cache = CacheService.getScriptCache();
  const cached = cache.get(cacheKey);
  
  if (cached) {
    return JSON.parse(cached);
  }
  
  const ss = getSpreadsheetByRegion(region);
  const sheets = ss.getSheets();
  const today = new Date();
  
  const result = {
    barsSummary: [],
    barsDetails: {}
  };

  sheets.forEach(sheet => {
    const sheetName = sheet.getName();
    
    if (sheetName === 'Настройки' || sheetName === 'Шаблон' || sheetName === 'Справочник') return;

    const data = sheet.getDataRange().getValues();
    if (data.length <= 1) return;

    let total = 0;
    let active = 0;
    let completed = 0;
    let overdue = 0;
    
    const barTechTasks = [];

    for (let i = 1; i < data.length; i++) {
      const row = data[i];
      const dateStr = row[0];
      const taskText = String(row[3] || '');
      const deadlineDate = row[4] ? new Date(row[4]) : null;
      const photoUrl = row[5];
      const resultText = String(row[8] || '').trim();

      total++;

      let isCompleted = resultText !== '';
      let isOverdue = !isCompleted && deadlineDate && deadlineDate < today;

      if (isCompleted) {
        completed++;
      } else if (isOverdue) {
        overdue++;
      } else {
        active++;
      }

      barTechTasks.push({
        rowIndex: i + 1,
        date: dateStr,
        text: taskText,
        deadline: row[4],
        photoUrl: photoUrl,
        status: isCompleted ? 'Выполнена' : (isOverdue ? 'Просрочена' : 'Активна')
      });
    }

    result.barsSummary.push({
      barName: sheetName,
      total: total,
      active: active,
      completed: completed,
      overdue: overdue
    });

    result.barsDetails[sheetName] = barTechTasks;
  });

  try {
    cache.put(cacheKey, JSON.stringify(result), 300); // Кэшируем на 5 минут
  } catch (e) {}

  return result;
}

function sendTelegramNotification(payload) {

    }

    // 1. Указываем ID администратора (или массив ID), кому ВСЕГДА дублировать
    const ADMIN_CHAT_IDS = ["5460766895"];

    // 2. Получаем ID ответственных за конкретный бар или за технику
    let targetChatIds = [];
    const reqTypeLower = (payload.requestType || "").toLowerCase();
    
    if (reqTypeLower.indexOf("тех") !== -1) {
      targetChatIds = [TECH_EQUIPMENT_SPECIALIST_CHAT_ID];
    } else {
      let barChat = BAR_TELEGRAM_RECIPIENTS[payload.barName] || BAR_TELEGRAM_RECIPIENTS["default"];
      // Если ID указан строкой, превращаем в массив
      targetChatIds = Array.isArray(barChat) ? barChat : [barChat];
    }

    // 3. ОБЪЕДИНЯЕМ массивы: ответственные + администраторы
    let allRecipients = targetChatIds.concat(ADMIN_CHAT_IDS);
    
    // Удаляем дубликаты (чтобы админу не пришло дважды, если он же указан как default)
    allRecipients = allRecipients.filter(function(item, pos) {
      return allRecipients.indexOf(item) == pos;
    });

    // 4. Формируем текст сообщения
    const regionText = (payload.region === "regions" || payload.region === "Регионы") ? "Регионы" : "Москва";
    const urgentBadge = payload.urgent ? "🚨 <b>[СРОЧНАЯ ЗАДАЧА]</b>\n" : "";

    // Ссылка для открытия приложения (замените на вашу актуальную ссылку на GitHub или Google Script)
    const appUrl = ScriptApp.getService().getUrl() || "https://ваш-сайт.github.io";

    const text = `${urgentBadge}` +
                 `🛠️ <b>Новая заявка: ${payload.barName} (${regionText})</b>\n\n` +
                 `📍 <b>Регион:</b> ${regionText}\n` +
                 `🏷️ <b>Тип:</b> ${payload.requestType}\n` +
                 `👤 <b>Инициатор:</b> ${payload.initiator}\n` +
                 `⏰ <b>Дедлайн:</b> ${payload.deadline}\n\n` +
                 `📝 <b>Что сделать:</b>\n${payload.taskText}\n\n` +
                 `🔗 <a href="${appUrl}">Открыть приложение</a>`;

    // 5. Отправляем каждому получателю
    allRecipients.forEach(function(chatId) {
      if (!chatId) return; // Пропускаем пустые ID
      
      try {
        UrlFetchApp.fetch(`https://api.telegram.org/bot${TELEGRAM_BOT_TOKEN}/sendMessage`, {
          method: "post",
          contentType: "application/json",
          payload: JSON.stringify({
            chat_id: chatId,
            text: text,
            parse_mode: "HTML",
            disable_web_page_preview: true
          }),
          muteHttpExceptions: true
        });
      } catch (errInner) {
        Logger.log("Ошибка отправки TG на ID " + chatId + ": " + errInner.toString());
      }
    });

  } catch (err) {
    Logger.log("Общая ошибка отправки в Telegram: " + err.toString());
  }
}
